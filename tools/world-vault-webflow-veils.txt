<script>
  // ─── Starfield ───
  (function () {
    const canvas = document.getElementById('stars');
    const ctx = canvas.getContext('2d');
    let stars = [];

    function resize() {
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
      const count = window.innerWidth < 560 ? 90 : 180;
      stars = [];
      for (let i = 0; i < count; i++) {
        stars.push({
          x: Math.random() * canvas.width,
          y: Math.random() * canvas.height,
          r: Math.random() * 0.8 + 0.2,
          a: Math.random() * 0.4 + 0.2,
          twinkle: Math.random() * Math.PI * 2,
          speed: Math.random() * 0.012 + 0.003,
        });
      }
    }

    function draw() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      const t = Date.now() * 0.001;
      stars.forEach(s => {
        const flicker = Math.sin(t * s.speed * 60 + s.twinkle) * 0.15 + 0.85;
        ctx.fillStyle = `rgba(220,228,240,${s.a * flicker})`;
        ctx.beginPath();
        ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
        ctx.fill();
      });
      requestAnimationFrame(draw);
    }

    resize();
    window.addEventListener('resize', resize);
    draw();

    // Wake the far-world
    setTimeout(() => {
      document.getElementById('farWorld').classList.add('awake');
    }, 100);
  })();

  // ─── Veil-specific copy ───
  const COPY = {

      veilPath: '/world-vault/mind',

    const params = new URLSearchParams(window.location.search);
    const veil = params.get('veil');

    if (!veil || !COPY[veil]) {
      // No veil specified — send them back to the threshold

    const c = COPY[veil];

      note.innerHTML = `This veil is already lifted on this device. <a href="${c.veilPath}">Continue →</a>`;
      note.style.display = 'block';
    }
  })();
